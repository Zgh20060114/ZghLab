  add_test(NAME ${test_name} COMMAND $<TARGET_FILE:${test_name}>)
  target_link_libraries(${test_name} PRIVATE tutorial_lib)
  target_precompile_headers(${test_name} REUSE_FROM tutorial_lib)
endmacro()
